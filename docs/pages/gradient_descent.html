§ I gradient_descent 

def gradient_descent(fn, params, lr, max_iters=1000):
    i = 0
    while True:
        new_params = addv(params, scalev(-lr, fn(params)))
        params = new_params
        i += 1
        if i >= max_iters:
            break
    return params
